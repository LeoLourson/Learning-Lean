```lean
theorem reduction_under_product_bound (a b κ u v f g : ℝ)
    (hκ : κ < 1) (hab : |a * b| ≤ κ) :
    (u + a * v = f ∧ b * u + v = g) ↔
      (u = (f - a * g) / (1 - a * b) ∧
       v = (g - b * f) / (1 - a * b)) := by
  sorry
```
